{{- /* Build a CF Images srcset for one output format. Returns a dict with the candidate list as srcset. CF Images returns the unconverted original when a width is not smaller than the original, so widths at or above maxWidth are left out. */ -}} {{- $entries := slice -}} {{- range .widths -}} {{- if or (eq $.maxWidth 0) (lt . $.maxWidth) -}} {{- $entries = $entries | append (printf "%s/width=%d,metadata=none,onerror=redirect,format=%s%s.%s %dw" $.baseURL . $.format $.path $.ext .) -}} {{- end -}} {{- end -}} {{- return dict "srcset" (delimit $entries ", ") -}}
